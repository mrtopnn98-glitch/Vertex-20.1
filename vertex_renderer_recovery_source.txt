1745: static ColorRGBA VxHue(float h)
1746: {
1747:         h -= std::floor(h);
1748:         const float f6 = h * 6.0f;
1749:         const int i = (int)f6 % 6;
1750:         const float f = f6 - std::floor(f6);
1751:         const float q = 1.0f - f, t = f;
1752:         switch(i)
1753:         {
1754:         case 0: return ColorRGBA(1.0f, t, 0.0f, 1.0f);
1755:         case 1: return ColorRGBA(q, 1.0f, 0.0f, 1.0f);
1756:         case 2: return ColorRGBA(0.0f, 1.0f, t, 1.0f);
1757:         case 3: return ColorRGBA(0.0f, q, 1.0f, 1.0f);
1758:         case 4: return ColorRGBA(t, 0.0f, 1.0f, 1.0f);
1759:         default: return ColorRGBA(1.0f, 0.0f, q, 1.0f);
1760:         }

1763: // pointed leaf: two freeform quads + optional darker midrib/stem. Must be called between QuadsBegin/QuadsEnd.
1764: static void VxDrawLeaf(IGraphics *pGraphics, float cx, float cy, float Rot, float Ln, float Wd, const ColorRGBA &Col, float Alpha, bool Rib)
1765: {
1766:         const float ca = std::cos(Rot), sa = std::sin(Rot);
1767:         const auto Pt = [&](float lx, float ly) { return vec2(cx + lx * ca - ly * sa, cy + lx * sa + ly * ca); };
1768:         const vec2 A = Pt(-Ln, 0.0f), B = Pt(-Ln * 0.35f, -Wd), C = Pt(Ln * 0.45f, -Wd * 0.8f), D = Pt(Ln, 0.0f);
1769:         const vec2 E = Pt(-Ln * 0.35f, Wd), F = Pt(Ln * 0.45f, Wd * 0.8f);
1770:         pGraphics->SetColor(ColorRGBA(Col.r, Col.g, Col.b, Alpha));
1771:         const IGraphics::CFreeformItem Up(A.x, A.y, B.x, B.y, D.x, D.y, C.x, C.y);
1772:         const IGraphics::CFreeformItem Lo(A.x, A.y, E.x, E.y, D.x, D.y, F.x, F.y);
1773:         pGraphics->QuadsDrawFreeform(&Up, 1);
1774:         pGraphics->QuadsDrawFreeform(&Lo, 1);
1775:         if(Rib)
1776:         {
1777:                 pGraphics->SetColor(ColorRGBA(Col.r * 0.6f, Col.g * 0.6f, Col.b * 0.6f, Alpha));
1778:                 const float Th = std::max(0.08f, Ln * 0.05f);
1779:                 const vec2 R0 = Pt(-Ln * 1.3f, -Th), R1 = Pt(Ln * 0.8f, -Th), R2 = Pt(-Ln * 1.3f, Th), R3 = Pt(Ln * 0.8f, Th);
1780:                 const IGraphics::CFreeformItem RibQ(R0.x, R0.y, R1.x, R1.y, R2.x, R2.y, R3.x, R3.y);
1781:                 pGraphics->QuadsDrawFreeform(&RibQ, 1);
1782:         }
1783: }

1933: // Rain: light / heavy, thin slanted streaks (screen space)
1934: static void VxRenderRain(IGraphics *pGraphics, float Dt, float W, float H)
1935: {
1936:         static std::vector<SVxDrop> s_vDrops;
1937:         const bool Heavy = g_Config.m_VxRainHeavy != 0;
1938:         const int Want = Heavy ? 520 : 140;
1939:         while((int)s_vDrops.size() < Want)
1940:         {
1941:                 SVxDrop D;
1942:                 D.m_X = VxRandF() * 1.4f - 0.2f;
1943:                 D.m_Y = VxRandF();
1944:                 D.m_Speed = Heavy ? 1.5f + VxRandF() * 0.8f : 0.8f + VxRandF() * 0.5f;
1945:                 D.m_Len = Heavy ? 0.035f + VxRandF() * 0.03f : 0.022f + VxRandF() * 0.02f;
1946:                 D.m_Alpha = 0.35f + VxRandF() * 0.4f;
1947:                 s_vDrops.push_back(D);
1948:         }
1949:         if((int)s_vDrops.size() > Want)
1950:                 s_vDrops.resize(Want);
1951: 
1952:         const float Slant = std::clamp(g_Config.m_VxRainSlant, 0, 100) / 100.0f * 0.45f; // dx/dy
1953:         const ColorRGBA Base = color_cast<ColorRGBA>(ColorHSLA(g_Config.m_VxRainColor));
1954:         const float Thick = std::max(0.25f, H * (Heavy ? 0.0016f : 0.0012f));
1955: 
1956:         pGraphics->TextureClear();
1957:         pGraphics->QuadsBegin();
1958:         for(SVxDrop &D : s_vDrops)
1959:         {
1960:                 D.m_Y += D.m_Speed * Dt;
1961:                 D.m_X += D.m_Speed * Slant * Dt * (H / W);
1962:                 if(D.m_Y > 1.05f)
1963:                 {
1964:                         D.m_Y = -D.m_Len;
1965:                         D.m_X = VxRandF() * 1.4f - 0.2f;
1966:                 }
1967:                 const float hx = D.m_X * W, hy = D.m_Y * H;
1968:                 const float tx = hx - D.m_Len * H * Slant, ty = hy - D.m_Len * H;
1969:                 const float dx = hx - tx, dy = hy - ty;
1970:                 const float Len = std::sqrt(dx * dx + dy * dy);
1971:                 if(Len < 0.001f)
1972:                         continue;
1973:                 const float nx = -dy / Len * Thick * 0.5f, ny = dx / Len * Thick * 0.5f;
1974:                 pGraphics->SetColor(ColorRGBA(Base.r, Base.g, Base.b, D.m_Alpha));
1975:                 const IGraphics::CFreeformItem Q(tx - nx, ty - ny, tx + nx, ty + ny, hx - nx, hy - ny, hx + nx, hy + ny);
1976:                 pGraphics->QuadsDrawFreeform(&Q, 1);
1977:         }
1978:         pGraphics->QuadsEnd();
1979: }

1996: // The screen must already be mapped to the game group (so units are map units, a tee is ~28 units wide).
1997: // Leaves fall with wind + sway + flip, stop on solid tiles (Collision()->CheckPoint), lie there for a while,

1999: void VertexRenderWorld(IGraphics *pGraphics, CCollision *pCollision, float Dt, vec2 PlayerPos, bool HasPlayer, float ViewX0, float ViewY0, float ViewX1, float ViewY1)
